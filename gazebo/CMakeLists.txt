cmake_minimum_required(VERSION 3.16)
project(urc_gz_system LANGUAGES CXX)

option(URC_GAZEBO_VENDOR "Link against ROS Jazzy's Gazebo vendor packages" OFF)
if(URC_GAZEBO_VENDOR)
    find_package(gz_sim_vendor REQUIRED)
    find_package(gz_transport_vendor REQUIRED)
    find_package(gz_msgs_vendor REQUIRED)
    set(sim gz-sim)
    set(transport gz-transport)
    set(msgs gz-msgs)
else()
    set(sim gz-sim8)
    set(transport gz-transport13)
    set(msgs gz-msgs10)
endif()
find_package(${sim} REQUIRED)
find_package(${transport} REQUIRED)
find_package(${msgs} REQUIRED)

add_library(urc_gz_system SHARED system.cpp)
target_compile_features(urc_gz_system PRIVATE cxx_std_17)
target_link_libraries(urc_gz_system PRIVATE ${sim}::${sim} ${transport}::core ${msgs}::core)
install(TARGETS urc_gz_system LIBRARY DESTINATION lib)
